cmake_minimum_required(VERSION 3.22)
project(land02_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(WORKER "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ABI "C:/Users/Matty/Dev/project-wroughtwild")
file(GLOB RULES CONFIGURE_DEPENDS "${WORKER}/sim/src/*.cpp")
add_library(land02_rules STATIC ${RULES})
target_include_directories(land02_rules PUBLIC "${WORKER}/sim/include")
target_compile_options(land02_rules PRIVATE -O2 -Wall -Wextra -fno-gnu-unique)
add_library(wroughtwild_sim SHARED "${WORKER}/game/extensions/wroughtwild_sim/src/register_types.cpp" "${WORKER}/game/extensions/wroughtwild_sim/src/wroughtwild_sim.cpp")
target_include_directories(wroughtwild_sim PRIVATE "${WORKER}/game/extensions/wroughtwild_sim/src" "${ABI}/third_party/godot-cpp/include" "${ABI}/build/gdext/godot-cpp/gen/include" "${ABI}/third_party/godot-cpp/gdextension")
target_compile_definitions(wroughtwild_sim PRIVATE DEBUG_ENABLED GDEXTENSION HOT_RELOAD_ENABLED THREADS_ENABLED WINDOWS_ENABLED)
target_compile_options(wroughtwild_sim PRIVATE -O2 -Wall -Wextra -fno-gnu-unique)
target_link_libraries(wroughtwild_sim PRIVATE land02_rules "${ABI}/build/gdext/bin/libgodot-cpp.windows.template_debug.x86_64.a" kernel32 user32 gdi32 winspool shell32 ole32 oleaut32 uuid comdlg32 advapi32)
target_link_options(wroughtwild_sim PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-undefined)
set_target_properties(wroughtwild_sim PROPERTIES PREFIX "lib" OUTPUT_NAME "wroughtwild_sim.windows.x86_64" RUNTIME_OUTPUT_DIRECTORY "${WORKER}/build/land02/native/bin")
