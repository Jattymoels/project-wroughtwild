var r2_mesh_digests:Dictionary={}
func r2_digest(mesh:Mesh)->String:
	var id:=mesh.get_instance_id()
	if r2_mesh_digests.has(id):return r2_mesh_digests[id]
	var hash:=HashingContext.new();hash.start(HashingContext.HASH_SHA256)
	for s in mesh.get_surface_count():hash.update(var_to_bytes(mesh.surface_get_arrays(s)))
	var result:=hash.finish().hex_encode();r2_mesh_digests[id]=result;return result
